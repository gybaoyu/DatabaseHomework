#[例3.74]  对每一个专业求学生的平均年龄，并把结果存入数据库
#第一步：建表
CREATE TABLE S_Age(
    Smajor VARCHAR(40),
    AvgAge DECIMAL(4,1)
);
         
 
 	

#第二步：插入数据
INSERT INTO S_Age(Smajor,AvgAge)
SELECT Smajor,AVG(TIMESTAMPDIFF(YEAR,Sbirthdate,CURDATE()))
FROM Student
GROUP BY Smajor;

SELECT * FROM S_Age;
