#6. 调整课程学分（2、2.5调为3；4调为4.5；其他加1.5）
UPDATE course
SET course_credit=
CASE
    WHEN course_credit IN (2.0,2.5) THEN 3.0
    WHEN course_credit=4.0 THEN 4.5
    ELSE course_credit+1.5
END;

#7. 在student表增加“选课门数”属性，并写入每个学生的选课数量
ALTER TABLE student
ADD COLUMN student_coursecount INT;

UPDATE student s
LEFT JOIN (
    SELECT student_id,COUNT(*) AS cnt
    FROM score
    GROUP BY student_id
) t ON s.student_id=t.student_id
SET s.student_coursecount=IFNULL(t.cnt,0);

#8. 计算每个班级中学生的平均年龄，并存入新表
CREATE TABLE class_avg_age(
    class_name VARCHAR(20),
    avg_age DECIMAL(4,1)
);

INSERT INTO class_avg_age(class_name,avg_age)
SELECT c.class_name,
       AVG(TIMESTAMPDIFF(YEAR,s.student_birthdate,CURDATE()))
FROM class c
JOIN student s ON c.class_id=s.class_id
GROUP BY c.class_name;

#9. 删除成绩低于所有课程平均成绩的选课记录
DELETE FROM score
WHERE score_grade IS NOT NULL
  AND score_grade<ALL(
      SELECT avg_grade
      FROM (
          SELECT AVG(score_grade) AS avg_grade
          FROM score
          WHERE score_grade IS NOT NULL
          GROUP BY course_id
      ) t
  );
