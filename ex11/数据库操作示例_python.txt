#1、使用pymysql连接MySQL服务器并创建数据库
import pymysql
try:
    #连接数据库，生成连接对象
    print("正在连接MySQL服务器......")
    connection = pymysql.connect(host='localhost',
                                 user='root',
                                 password='root',
                                 charset='utf8mb4')
    
    #创建游标对象
    cursor = connection.cursor()
    print("连接MySQL服务器成功！")
    #创建数据库
    database_name='mydatabase'
    cursor.execute(f"SHOW DATABASES LIKE '{database_name}'")
    results=cursor.fetchone()
    if results:
        print(f"数据库'{database_name}'已存在")
        #可删除已有数据库再重新创建
        #if cursor.execute(f"DROP DATABASE {database_name}"):
        #    print(f"已删除旧数据库'{database_name}'")
    else:
        #创建数据库
        create_sql=f"""CREATE DATABASE {database_name} CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci ;"""
        if cursor.execute(create_sql):
            print(f"数据库'{database_name}'创建成功")
except pymysql.Error as e:
    print(f"数据库操作出错：{e}")

            CREATE TABLE IF NOT EXISTS student (

                Sbirth DATE COMMENT '出生日期',
                Sdept VARCHAR(30) COMMENT '所在院系'

        """
sql_create_course = f"""
            CREATE TABLE IF NOT EXISTS course (
                Cno CHAR(5) PRIMARY KEY COMMENT '课程号',

                Cpno CHAR(5) COMMENT '先修课程号',
                FOREIGN KEY (Cpno) REFERENCES course(Cno) ON DELETE SET NULL

        """
