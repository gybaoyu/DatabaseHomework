#1. 查询 student 表中所有学生的详细信息。
select * from student;
#2. 查询所有开设课程的课程名称及考核方式。
select course_name,course_assessmentmethod from course;
#3. 查询所有选课学生的学号（如一个同学同时选修了多门课程，该学号只显示一次）。
select distinct student_id from score;
#4. 查询所有有学生选修的课程编号(一门课程有多名同学选修，该课程编号只显示一次)。
select distinct course_id from score;
#5. 查询所有学时在[40,60]范围内的课程编号和课程名称(分别用 AND 和 BETWEEN 运算符实现)。
select course_id,course_name from course where course_hours>=40 and course_hours<=60;
select course_id,course_name from course where course_hours between 40 and 60;
#6. 查询所有学生党员的学号和姓名。
select student_id,student_name from student where student_partymember=0x01;
#7. 查询所有姓“张”的学生的姓名和性别(分别用 LEFT 函数和 LIKE 运算符实现)。（参见附例：LEFT 函数）
select student_name,student_sex from student where left(student_name,1)='张';
select student_name, student_sex from student where student_name like '张%';
#8. 查询所有学生的姓名及年龄，并按照年龄从小到大的顺序显示，查询结果中列名为姓名、年龄。
select student_name,student_birthdate from student order by student_birthdate;
#9. (*)查询入学成绩在前 20%的学生姓名，最后一名如有并列，只显示一个。
SELECT student_name
FROM (
    SELECT *,
           ROW_NUMBER() OVER (ORDER BY student_entrancescore DESC) AS row_num,
           COUNT(*) OVER () AS total_count
    FROM student
) AS ranked
WHERE row_num <= total_count * 0.2;
#10. 查询入学成绩排在前 3 位的学生姓名，最后一名如有并列则全部显示。（参见附例：LIMIT关键字）
select student_name from student order by student_entrancescore desc limit 3;
#11. 查询所有 10 月份出生的学生人数。
SELECT COUNT(*) AS count FROM student WHERE MONTH(student_birthdate) = 10;
#12. 查询没有先修课的课程名称。
select course_name from course where course_prerequisite is null;
#13. 查询入学成绩的最高分和最低分，查询结果中列名分别为最高分和最低分。
select max(student_entrancescore) as 最高分,min(student_entrancescore) as 最低分 from student;
#14. 查询每门选修课程成绩的最高分和最低分，查询结果中列名分别为课程编号、最高分和最低分。
select course_id as 课程编号,max(score_grade)as 最高分,min(score_grade)as 最低分 from score group by course_id;
#15. 统计男女生的入学平均成绩，查询结果中显示性别和平均成绩两列。
select student_sex as 性别, avg(student_entrancescore) as 平均成绩 from student group by student_sex;
#16. 统计考试和考查两种性质课程的总学分，查询结果中显示课程性质和总学分两列。
select course_assessmentmethod as 课程性质,sum(course_credit) as 总学分 from course group by course_assessmentmethod;
#17. 查询选修了两门以上(包括两门)课程的学生的学号。
select student_id from score group by student_id having count(distinct course_id) >= 2;
#18. 查询所有第 2 学期开课的选课信息。（使用 RIGHT 函数，与 LEFT 函数类似）
select distinct course_id from score where right(score_semester,'2');
select course_id,course_name from course where course_id in (select distinct course_id from score where right(score_semester,'2'));
#19. 查询当前时刻。（curdate()）
select curdate() as 当前时刻;
#20. 从 student 中随机查询 3 名学生的详细信息。(order by rand())
select * from student order by rand() limit 3;
