#[例3.19] 查全体学生的姓名及其年龄（表中只有出生日期）

    student_name,
    YEAR(CURDATE()) - YEAR(student_birthdate) -

FROM student;

#[例3.20] 查询全体学生的姓名、出生日期和主修专业
#要求用小写字母表示主修专业

    s.student_name,
    s.student_birthdate,
    LOWER(c.class_major) AS 主修专业
FROM student s
JOIN class c ON s.class_id = c.class_id;

#[例3.25] 查询年龄在20~23岁（包括20岁和23岁）之间的学生的姓名、出生日期和主修专业
SELECT 
    s.student_name,
    s.student_birthdate,
    c.class_major
FROM student s
JOIN class c ON s.class_id = c.class_id
WHERE (YEAR(CURDATE()) - YEAR(student_birthdate) - 
      (DATE_FORMAT(CURDATE(), '%m%d') < DATE_FORMAT(student_birthdate, '%m%d'))) BETWEEN 20 AND 23;

#[例3.26] 查询年龄不在20~23岁之间的学生的姓名、出生日期和主修专业

FROM score
ORDER BY course_id ASC, score_grade DESC;

#【例】 查询有两门以上不及格课程的学生学号及其平均成绩
SELECT student_id, AVG(score_grade) AS 平均成绩
FROM score
WHERE student_id IN (
    SELECT student_id
    FROM score
    WHERE score_grade < 60
    GROUP BY student_id
    HAVING COUNT(*) > 1
)
AND score_grade IS NOT NULL
GROUP BY student_id;

#[例3.49] 查询选修了数据库系统概论课程的成绩排名前10名的学生学号
SELECT s.student_id
FROM score sc
JOIN course c ON sc.course_id = c.course_id
WHERE c.course_name = '数据库系统概论' AND sc.score_grade IS NOT NULL
ORDER BY sc.score_grade DESC
LIMIT 10;

#[例3.50] 查询平均成绩排名在3~5名的学生学号和平均成绩

FROM score
WHERE score_grade IS NOT NULL
GROUP BY student_id
ORDER BY AVG(score_grade) DESC
LIMIT 2, 3;  -- 从第3名开始，取3条记录（即3-5名）
