CREATE TABLE class(
	班级编号 char(6) PRIMARY KEY,
	班级名称 varchar(20) NOT NULL,
	所属专业 varchar(20) NOT NULL,
	班级人数 int NULL,
	CONSTRAINT ck_class_rs CHECK  (班级人数>=20 AND 班级人数<=40)
);

CREATE TABLE course(
	课程编号 char(8) NOT NULL PRIMARY KEY,
	课程名称 varchar(40) NOT NULL,
	考核方式 char(4) NOT NULL DEFAULT '考试' ,
	学时 int NOT NULL ,
	学分 DECIMAL(4, 1) NOT NULL,
	先修课 char(8) NULL,
	CONSTRAINT ck_course_khfs CHECK  ((考核方式='考试' OR 考核方式='考查')),
	CONSTRAINT ck_course_xs CHECK  ((学时>=(30) AND 学时<=(80)))
);

CREATE TABLE student(
	学号 char(10) NOT NULL PRIMARY KEY,
	姓名 varchar(8) NOT NULL,
	性别 char(2) NOT NULL DEFAULT '男' ,
	出生日期 date NOT NULL,
	入学成绩 DECIMAL(4, 1) NOT NULL ,
	党员否 bit NOT NULL,
	简历 text NULL,
	照片 BLOB NULL,
	班级编号 char(6) NOT NULL,
	CONSTRAINT ck_student_xb CHECK  ((性别='男' OR 性别='女')),
	CONSTRAINT ck_student_rxcj CHECK  ((入学成绩>=(350) AND 入学成绩<=(750))),
	CONSTRAINT fk_student_class FOREIGN KEY(班级编号) REFERENCES class (班级编号)
); 
 
CREATE TABLE score(
	学号 char(10) NOT NULL ,
	课程编号 char(8) NOT NULL ,
	成绩 DECIMAL(4, 1) NULL ,
	学期 char(9) NOT NULL,
   PRIMARY KEY (学号 ,课程编号),
   CONSTRAINT fk_score_student FOREIGN KEY (学号) REFERENCES student (学号),
   CONSTRAINT fk_score_course FOREIGN KEY (课程编号) REFERENCES course(课程编号),
   CONSTRAINT ck_score_cj CHECK  ((成绩>=(0) AND 成绩<=(100)))
);
