-- 1.(1) 建立所有男同学的视图 VBoy_Student、所有女同学的视图 VGirl_Student
-- 视图中包括学生表中的所有列
DROP VIEW IF EXISTS VBoy_Student;
CREATE VIEW VBoy_Student AS
SELECT * FROM student WHERE student_sex = '男';

DROP VIEW IF EXISTS VGirl_Student;
CREATE VIEW VGirl_Student AS
SELECT * FROM student WHERE student_sex = '女';


-- 1.(2) 建立国际贸易081班选修了04010101号课程且成绩在60分以上的学生的视图
-- 视图中包括学生姓名、课程编号和成绩
DROP VIEW IF EXISTS Vgjmv081good_04010101;
CREATE VIEW Vgjmv081good_04010101 AS
SELECT 
    s.student_name,
    sc.course_id,
    sc.score_grade
FROM student s
JOIN score sc ON s.student_id = sc.student_id
JOIN class c ON s.class_id = c.class_id
WHERE c.class_name = '国际贸易081'
  AND sc.course_id = '04010101'
  AND sc.score_grade >= 60;


-- 1.(3) 建立一个反映所有学生姓名和年龄的视图 VS_AGE
DROP VIEW IF EXISTS VS_AGE;
CREATE VIEW VS_AGE AS
SELECT 
    student_name,
    TIMESTAMPDIFF(YEAR, student_birthdate, CURDATE()) AS age
FROM student;


-- 1.(4) 将学生的学号及该生的平均成绩定义为一个视图 Vpjcj_Student
DROP VIEW IF EXISTS Vpjcj_Student;
CREATE VIEW Vpjcj_Student AS
SELECT 
    student_id,
    AVG(score_grade) AS avg_score
FROM score
GROUP BY student_id;


-- 1.(5) 将课程编号、课程名称及选修人数定义为一个视图 VCount_Xuanxiu
DROP VIEW IF EXISTS VCount_Xuanxiu;
CREATE VIEW VCount_Xuanxiu AS
SELECT 
    c.course_id,
    c.course_name,
    COUNT(sc.student_id) AS cnt
FROM course c
LEFT JOIN score sc ON c.course_id = sc.course_id
GROUP BY c.course_id, c.course_name;


-- 1.(6) 创建反映国际贸易081班男生人数和女生人数的一个视图 Vboygirlnumber
DROP VIEW IF EXISTS Vboygirlnumber;
CREATE VIEW Vboygirlnumber AS
SELECT '男' AS 性别, COUNT(*) AS 人数
FROM student s
JOIN class c ON s.class_id = c.class_id
WHERE c.class_name = '国际贸易081' AND s.student_sex = '男'

UNION ALL

SELECT '女', COUNT(*)
FROM student s
JOIN class c ON s.class_id = c.class_id
WHERE c.class_name = '国际贸易081' AND s.student_sex = '女'

UNION ALL

SELECT '合计', COUNT(*)
FROM student s
JOIN class c ON s.class_id = c.class_id
WHERE c.class_name = '国际贸易081';


-- 1.(7) 创建视图"分段统计"，显示为每门课程良好以上及以下的学生人数
-- 良好分数线为80分（良好以上：score_grade >= 80，良好以下：score_grade < 80）
DROP VIEW IF EXISTS 分段统计;
CREATE VIEW 分段统计 AS
SELECT 
    sc.course_id AS 课程编号,
    SUM(CASE WHEN sc.score_grade >= 80 THEN 1 ELSE 0 END) AS 良好以上,
    SUM(CASE WHEN sc.score_grade < 80 THEN 1 ELSE 0 END) AS 良好以下,
    COUNT(sc.student_id) AS 合计
FROM score sc
GROUP BY sc.course_id

UNION ALL

SELECT 
    c.course_id,
    0,
    0,
    0
FROM course c
WHERE c.course_id NOT IN (SELECT DISTINCT course_id FROM score);


-- 2.(1) 在所有男同学的视图 VBoy_Student 中找出年龄小于21岁的学生

-- 2.(2) 在所有学生年龄的视图 VS_AGE 中查询比"张楚"年龄还小的学生
SELECT * FROM VS_AGE
WHERE age < (SELECT age FROM VS_AGE WHERE student_name = '张楚');


-- 2.(3) 在视图 Vpjcj_Student 查询平均成绩小于60的学生的学号和平均成绩
SELECT * FROM Vpjcj_Student
WHERE avg_score < 60;


-- 2.(4) 在 VCount_Xuanxiu 中查询选修人数在2人以上的课程编号
SELECT course_id FROM VCount_Xuanxiu
WHERE cnt >= 2;
