#第3章视图语句练习
#1、创建视图
#[例3.86]  建立信息管理与信息系统专业学生的视图IS_student
CREATE VIEW IS_Student
AS
SELECT Sno,Sname,Ssex,Sbirthdate,Smajor
FROM Student
WHERE Smajor='信息管理与信息系统';

SELECT * FROM IS_Student;

#[例3.87] 建立信息管理与信息系统专业学生的视图IS_student_Check，要求进行修改和插入操作时仍需保证该视图只有信息管理与信息系统专业的学生
CREATE VIEW IS_Student_Check
AS
SELECT Sno,Sname,Ssex,Sbirthdate,Smajor
FROM Student
WHERE Smajor='信息管理与信息系统'
WITH CHECK OPTION;

SELECT * FROM IS_Student_Check;

#基于多个基表的视图
#[例3.88]  建立信息管理与信息系统专业选修了81001号课程的学生的视图IS_C1（包括学号、姓名、成绩）
CREATE VIEW IS_C1(Sno,Sname,Grade)
AS
SELECT Student.Sno,Sname,Grade
FROM Student,SC
WHERE Smajor='信息管理与信息系统'
  AND Student.Sno=SC.Sno
  AND SC.Cno='81001';

SELECT * FROM IS_C1;

#基于视图的视图
#[例3.89]  建立信息管理与信息系统专业选修了81001号课程且成绩在90分以上的学生的视图IS_C2
CREATE VIEW IS_C2
AS
SELECT *
FROM IS_C1
WHERE Grade>=90;

SELECT * FROM IS_C2;

#带表达式的视图
#[例3.90]  定义一个反映学生年龄的视图S_AGE
CREATE VIEW S_AGE(Sno,Sname,Sage)
AS
SELECT Sno,Sname,TIMESTAMPDIFF(YEAR,Sbirthdate,CURDATE())
FROM Student;

SELECT * FROM S_AGE;

#分组视图
#[例3.91]  将学生的学号及平均成绩定义为一个视图S_GradeAVG
CREATE VIEW S_GradeAVG(Sno,Gavg)
AS
SELECT Sno,AVG(Grade)
FROM SC
GROUP BY Sno;

SELECT * FROM S_GradeAVG;

#[例3.92] 将Student表中所有女生记录定义为一个视图F_Student
