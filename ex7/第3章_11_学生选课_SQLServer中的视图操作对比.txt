--2、在SQLServer中，删除视图is_s1,在is_s1上有定义的有视图is_s2
CREATE VIEW IS_S1(Sno,Sname,Grade)
AS
SELECT Student.Sno,Sname,Grade
FROM Student,SC
WHERE Sdept= 'CS' 
		AND Student.Sno=SC.Sno 
		AND SC.Cno= '1';
go
SELECT * FROM IS_S1;
go
CREATE VIEW IS_S2 
AS
SELECT Sno,Sname,Grade
FROM IS_S1
WHERE Grade>=90;
go
SELECT * FROM is_s2;
go

--下面语句，在SQLServer中，语法有误；在MySQL中，可执行，但其中的cascade功能被忽略
drop view is_s1 --cascade;

drop table student;
go
